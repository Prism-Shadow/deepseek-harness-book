\documentclass[tikz,border=5pt]{standalone}
\usepackage{fontspec}
\usepackage{tikz}
\usetikzlibrary{arrows.meta,backgrounds,positioning}

\setmainfont{Noto Sans CJK SC}
\definecolor{accent}{HTML}{315EF5}
\definecolor{light}{HTML}{EEF2FF}
\definecolor{textcolor}{HTML}{273142}

\begin{document}
\begin{tikzpicture}[
  show background rectangle,
  background rectangle/.style={fill=white,draw=none},
  node distance=12mm,
  flow/.style={
    draw=accent,
    fill=light,
    text=textcolor,
    minimum width=32mm,
    minimum height=11mm,
    align=center,
    font=\fontsize{11}{13}\selectfont
  },
  final/.style={
    flow,
    fill=accent,
    text=white
  },
  arrow/.style={
    draw=accent,
    line width=0.8pt,
    -{Latex[length=2.6mm,width=1.8mm]}
  }
]
  \node[flow] (act) {执行任务};
  \node[flow, right=of act] (feedback) {获得反馈};
  \node[flow, right=of feedback] (analyse) {分析问题};
  \node[final, right=of analyse] (change) {修改自身};

  \draw[arrow] (act) -- (feedback);
  \draw[arrow] (feedback) -- (analyse);
  \draw[arrow] (analyse) -- (change);
\end{tikzpicture}
\end{document}
