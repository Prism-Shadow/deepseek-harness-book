% Planned outline
\newcommand{\DSHPlanPage}[1]{%
  \ifstrempty{#1}{}{\hyperref[#1]{\pageref*{#1}}}%
}
\newenvironment{hhplannedoutline}{%
  \cleardoublepage
  \thispagestyle{plain}
  \phantomsection
  \begin{center}
    {\sffamily\DSHHeadingFont\bfseries\fontsize{20pt}{26pt}\selectfont\color{DSHInk}目录}
  \end{center}
  \vspace{7mm}
  \setlength{\parindent}{0pt}
  \setlength{\parskip}{0pt}
}{%
  \clearpage
}
